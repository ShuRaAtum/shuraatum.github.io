{%- comment -%} Links to the same page in every site language by swapping the first path segment (/en, /ko, /ja). The current language is shown but not linked. {%- endcomment -%} {%- assign current_lang = page.lang | default: "en" -%} {%- assign segments = page.url | replace_first: "/" | split: "/" -%} {%- assign rest = "" -%} {%- for seg in segments offset:1 -%} {%- assign rest = rest | append: "/" | append: seg -%} {%- endfor -%} {%- assign trailing = page.url | slice: -1 -%} {%- assign languages = "en,ko,ja" | split: "," -%}
{%- for code in languages -%} {%- assign code_strings_key = "strings_" | append: code -%} {%- assign native = site.data[code_strings_key].lang.native -%} {%- if page.lang -%} {%- assign target = "/" | append: code | append: rest -%} {%- if trailing == "/" -%}{%- assign target = target | append: "/" -%}{%- endif -%} {%- else -%} {%- comment -%} Pages without a language have no translations; go to that language's home. {%- endcomment -%} {%- assign target = "/" | append: code | append: "/" -%} {%- endif -%} {%- if page.lang and code == current_lang -%} {{ code | upcase }} {%- else -%} {{ code | upcase }} {%- endif -%} {%- endfor -%}
